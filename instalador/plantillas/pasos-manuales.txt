
# 2. GitHub, con la cuenta del dueño ({{dueno}}).
# 2a. Invita a {{bot}} como colaborador con escritura.
gh api -X PUT repos/{{dueno}}/{{repo}}/collaborators/{{bot}} -f permission=push
# 2b. Acepta la invitación con la configuración de gh del bot: el primer comando busca el id de la invitación y el segundo la acepta.
id=$(GH_CONFIG_DIR="$HOME/.talos-gh" gh api user/repository_invitations --jq '.[] | select((.repository.full_name | ascii_downcase) == "{{repo_completo_min}}") | .id'); echo "invitación: $id"
GH_CONFIG_DIR="$HOME/.talos-gh" gh api -X PATCH user/repository_invitations/$id
# 2c. Protección de main y staging. Aplícala después de que el CI corrió una vez en las dos ramas, para que existan los checks secrets y hooks.
#     Si el repo también corre el job node o python, agrégalo a contexts y a checks con el mismo app_id.
gh api -X PUT repos/{{dueno}}/{{repo}}/branches/main/protection --input - <<'JSON'
{
  "required_status_checks": {"strict": false, "contexts": ["secrets", "hooks"]},
  "enforce_admins": true,
  "required_pull_request_reviews": {
    "required_approving_review_count": 1,
    "require_code_owner_reviews": true,
    "dismiss_stale_reviews": true
  },
  "restrictions": null,
  "allow_force_pushes": false,
  "allow_deletions": false
}
JSON
# El PATCH fija los checks secrets y hooks a la app de GitHub Actions (app_id 15368).
gh api -X PATCH repos/{{dueno}}/{{repo}}/branches/main/protection/required_status_checks --input - <<'JSON'
{"strict": false, "checks": [{"context": "secrets", "app_id": 15368}, {"context": "hooks", "app_id": 15368}]}
JSON
gh api -X PUT repos/{{dueno}}/{{repo}}/branches/staging/protection --input - <<'JSON'
{
  "required_status_checks": {"strict": false, "contexts": ["secrets", "hooks"]},
  "enforce_admins": true,
  "required_pull_request_reviews": {
    "required_approving_review_count": 0,
    "require_code_owner_reviews": true,
    "dismiss_stale_reviews": false
  },
  "restrictions": null,
  "allow_force_pushes": false,
  "allow_deletions": false
}
JSON
# El PATCH fija los checks secrets y hooks a la app de GitHub Actions (app_id 15368).
gh api -X PATCH repos/{{dueno}}/{{repo}}/branches/staging/protection/required_status_checks --input - <<'JSON'
{"strict": false, "checks": [{"context": "secrets", "app_id": 15368}, {"context": "hooks", "app_id": 15368}]}
JSON
# 2d. Verifica la protección de las dos ramas. Esperado: main con 1, true, true, true, false, false;
#     staging con 0, true, false, true, false, false; en las dos, checks secrets:15368 y hooks:15368.
filtro='{aprobaciones: .required_pull_request_reviews.required_approving_review_count, code_owners: .required_pull_request_reviews.require_code_owner_reviews, descartar_viejas: .required_pull_request_reviews.dismiss_stale_reviews, enforce_admins: .enforce_admins.enabled, force_push: .allow_force_pushes.enabled, borrado: .allow_deletions.enabled, checks: [.required_status_checks.checks[] | "\(.context):\(.app_id)"]}'
gh api repos/{{dueno}}/{{repo}}/branches/main/protection --jq "$filtro"
gh api repos/{{dueno}}/{{repo}}/branches/staging/protection --jq "$filtro"
